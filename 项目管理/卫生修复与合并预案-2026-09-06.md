### 3.3 一键清理脚本（库外，审阅后自行执行）

```powershell
# cleanup_dups_20260906.ps1
# 删除 975 个 __dup 文件 + 1 个空白 Excalidraw 画布
# 用法:  先演练  powershell -File cleanup_dups_20260906.ps1
#        真删除  把下面 $WhatIf 保持 $true 改为 $false
$WhatIf = $true
$root = 'D:\The Memory\The Memory'
Get-ChildItem -Path $root -Recurse -File -Filter '*__dup*' |
  Where-Object { $_.FullName -notmatch '\\\.git\\' } |
  ForEach-Object {
    $orig = $_.FullName -replace '__dup\d+',''
    if (Test-Path $orig) {
      $a=(Get-FileHash $_.FullName -Algorithm MD5).Hash
      $b=(Get-FileHash $orig     -Algorithm MD5).Hash
      if ($a -eq $b) { Remove-Item $_.FullName -Confirm:$false -WhatIf:$WhatIf }
      else { Write-Host "跳过(内容不同): $($_.Name)" }
    } else { Write-Host "跳过(无原件): $($_.Name)" }
  }
# 空白画布二选一（保留较早的 08-10，删除 08-31）
Remove-Item 'D:\The Memory\The Memory\Excalidraw\Drawing 2026-08-31 16.49.47.excalidraw.md' -Confirm:$false -WhatIf:$WhatIf
```
